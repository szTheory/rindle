path_identity() {
  local path="$1"
  if [ ! -e "$path" ]; then
    printf 'ABSENT\n'
  elif [ -f "$path" ]; then
    printf 'FILE:%s\n' "$(shasum -a 256 "$path" | awk '{print $1}')"
  elif [ -d "$path" ]; then
    (
      cd "$path"
      find . -type f -print0 | LC_ALL=C sort -z | while IFS= read -r -d '' entry; do
        printf '%s:%s\n' "$entry" "$(shasum -a 256 "$entry" | awk '{print $1}')"
      done
    ) | shasum -a 256 | awk '{print "DIR:" $1}'
  else
    printf 'OTHER:%s\n' "$(stat -f '%HT:%Sp' "$path")"
  fi
}
